# NS-BRKGA Ablation Study — Stage 1: Vanilla NS-BRKGA
#
# Single population, fixed elite size (min == max), no diversity,
# no exchange, no path relinking, no shaking, no reset.
#
# The synthetic parameter "elites_percentage" is mapped by the ablation
# runner to both --min-elites-percentage and --max-elites-percentage.
#
# Focused retuning (round 2): each range is restricted to the region held by
# the round-1 surviving ROULETTE configurations on this problem (p10-p90 band
# where they were concentrated; interquartile band, extended to the a-priori
# value, where they were diffuse). Crossover is fixed to ROULETTE and path
# relinking (stages 4-6) to the binary-search IPR.

population_size_factor   "--population-size-factor "    i   (75, 125)
elites_percentage        "--elites-percentage "         r   (0.30, 0.45)
mutation_probability     "--mutation-probability "      r   (0.01, 0.10)
mutation_distribution    "--mutation-distribution "     r   (20.00, 50.00)
num_total_parents        "--num-total-parents "         i   (2, 4)
num_elite_parents        "--num-elite-parents "         i   (1, 3)
# CONSTANT=0, CUBIC=1, EXPONENTIAL=2, LINEAR=3, LOGINVERSE=4, QUADRATIC=5, SQRT=6
# Restricted to the mild-bias regime kept in round 1 (CONSTANT, LOGINVERSE, SQRT)
bias_type                "--bias-type "                 c   (0, 4, 6)
# ROULETTE=0, GEOMETRIC=1
# Fixed: ROULETTE only (focused retuning)
crossover_type           "--crossover-type "            c   (0)
# Fixed: diversity_type=0 (NONE), num_populations=1, all intervals=0
diversity_type           "--diversity-type "            c   (0)
num_populations          "--num-populations "           c   (1)
exchange_interval        "--exchange-interval "         c   (0)
num_exchange_individuals "--num-exchange-individuals "  c   (1)
pr_interval              "--pr-interval "               c   (0)
shake_interval           "--shake-interval "            c   (0)
reset_interval           "--reset-interval "            c   (0)

[global]
digits = 2

[forbidden]
(num_elite_parents > num_total_parents)
(num_elite_parents > (population_size_factor * 4) * elites_percentage)
